<script>
let state = JSON.parse(localStorage.getItem("po_state_v2")) || {
compliance:50, tension:30, fatigue:10, audit:20,
busyUntil:0, bell:null
};

function save(){localStorage.setItem("po_state_v2",JSON.stringify(state));}

function log(msg){
document.getElementById("log").innerHTML="• "+msg+"<br>"+document.getElementById("log").innerHTML;
}

function updateUI(){
document.getElementById("compliance").innerText=state.compliance;
document.getElementById("tension").innerText=state.tension;
document.getElementById("fatigue").innerText=state.fatigue;
document.getElementById("audit").innerText=state.audit;

let busy = Date.now()<state.busyUntil;
document.querySelectorAll("button").forEach(b=>b.disabled=busy && !state.bell);
document.getElementById("status").innerText=busy?"On duty…":"Idle";

if(state.bell){
document.getElementById("bellPanel").style.display="block";
let left = Math.max(0,Math.floor((state.bell.deadline-Date.now())/1000));
document.getElementById("bellTimer").innerText=left+"s";
if(left<=0){autoFailBell();}
}else{
document.getElementById("bellPanel").style.display="none";
}
save();
}

function startAction(name,min){
state.busyUntil=Date.now()+min*60000;
state.fatigue+=3;
state.compliance+=2;
log(name+" in progress.");
}

function triggerBell(){
let roll=Math.random();
let type="nonsense";
if(roll<0.01) type="real";
else if(roll<0.25) type="manipulation";

state.bell={
type:type,
deadline:Date.now()+300000
};
log("Cell bell activated.");
}

function resolveBell(outcome){
if(state.bell.type==="real"){
if(outcome==="respond"){
log("REAL INCIDENT handled successfully.");
state.compliance+=5;
state.tension-=5;
}else{
log("REAL INCIDENT escalated. Investigation pending.");
state.audit+=10;
state.tension+=10;
}
}else{
log("Cell bell was a false / minor issue.");
}
state.bell=null;
}

function respondBell(){
state.fatigue+=5;
resolveBell("respond");
}

function delayBell(){
resolveBell("delay");
}

function ignoreBell(){
state.audit+=5;
resolveBell("ignore");
}

function autoFailBell(){
log("No response to cell bell logged.");
state.audit+=10;
resolveBell("ignore");
}

setInterval(()=>{
if(!state.bell && Math.random()<0.02) triggerBell();
updateUI();
},1000);

updateUI();
</script>
